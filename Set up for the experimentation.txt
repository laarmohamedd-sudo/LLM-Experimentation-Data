function personaFor(i: number) {
  return PERSONAS[i % PERSONAS.length];
}

function flavorFor(i: number) {
  return FLAVORS[i % FLAVORS.length];
}

function tempFor(i: number) {
  const base = 0.9 + Math.random() * 0.1;
  const jitter = (i % 7) * 0.001;
  return Math.min(1, +(base + jitter).toFixed(3));
}

function sleep(ms: number) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function cleanJson(text: string) {
  return text.replace(/```json/gi, "").replace(/```/g, "").trim();
}

function parseJSON<T>(text: string): T | null {
  try {
    const cleaned = cleanJson(text);
    const start = cleaned.indexOf("{");
    const end = cleaned.lastIndexOf("}");
    const slice = start >= 0 && end > start ? cleaned.slice(start, end + 1) : cleaned;
    return JSON.parse(slice) as T;
  } catch {
    return null;
  }
}

function mean(xs: number[]) {
  const arr = xs.filter((x) => Number.isFinite(x));
  if (!arr.length) return NaN;
  return arr.reduce((a, b) => a + b, 0) / arr.length;
}

function pct(num: number, den: number) {
  if (!den) return 0;
  return +(100 * num / den).toFixed(1);
}

function qreProbability(ua: number, ub: number, lambda: number) {
  return 1 / (1 + Math.exp(-lambda * (ua - ub)));
}

function loadSnapshot(): Snapshot | null {
  try {
    for (const key of STORAGE_KEYS) {
      const raw = localStorage.getItem(key);
      if (raw) return JSON.parse(raw) as Snapshot;
    }
    return null;
  } catch {
    return null;
  }
}

function saveSnapshot(payload: Snapshot) {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
  } catch {
    // ignore
  }
}

function normalizeRows(rows: RawRow[]) {
  const map = new Map<string, RawRow>();
  for (const row of rows) {
    const key = `${row.experimentId}::${row.condition}::${row.iteration}`;
    map.set(key, row);
  }
  return Array.from(map.values()).sort((a, b) => {
    if (a.experimentId !== b.experimentId) return a.experimentId.localeCompare(b.experimentId);
    if (a.condition !== b.condition) return a.condition.localeCompare(b.condition);
    return a.iteration - b.iteration;
  });
}

function bestResponsePDAction() {
  return "D";
}

function bestResponseBertrand(mc: number, beliefPrice: number) {
  const grid = Array.from({ length: 21 }, (_, i) => mc + i);
  let bestP = grid[0];
  let bestProfit = -Infinity;

  for (const p of grid) {
    let profit = 0;
    if (p < mc) profit = -Infinity;
    else if (p < beliefPrice) profit = p - mc;
    else if (p === beliefPrice) profit = 0.5 * (p - mc);
    else profit = 0;

    if (profit > bestProfit || (profit === bestProfit && p < bestP)) {
      bestProfit = profit;
      bestP = p;
    }
  }

  return bestP;
}

function bestResponseAuction(v: number, beliefHighBid: number) {
  const grid = Array.from({ length: v + 1 }, (_, i) => i);
  let bestB = grid[0];
  let bestU = -Infinity;

  for (const b of grid) {
    let u = 0;
    if (b > v) u = -Infinity;
    else if (b > beliefHighBid) u = v - b;
    else if (b === beliefHighBid) u = 0.5 * (v - b);
    else u = 0;

    if (u > bestU || (u === bestU && b < bestB)) {
      bestU = u;
      bestB = b;
    }
  }

  return bestB;
}

function brMix(role: "row" | "col", belief: number) {
  if (role === "row") return belief > 1 / 3 ? "U" : "D";
  return belief > 2 / 3 ? "L" : "R";
}

async function callLLM(
  prompt: string,
  temperature: number,
  shouldStop?: () => boolean
) {
  while (true) {
    if (shouldStop?.()) throw new Error("Stopped");

    try {
      const res = await fetch("http://localhost:3001/api/experiments/call", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ prompt, temperature }),
      });

      if (res.ok) {
        const data = (await res.json()) as { text?: string };
        return String(data.text ?? "");
      }

      if (res.status === 429) {
        const retryAfter = res.headers.get("retry-after");
        const waitMs = retryAfter ? Math.max(5000, Number(retryAfter) * 1000) : 5000;
        await sleep(waitMs);
        continue;
      }

      if (res.status >= 500) {
        await sleep(5000);
        continue;
      }

      throw new Error(`API ${res.status}: ${await res.text()}`);
    } catch (err) {
      if (shouldStop?.()) throw new Error("Stopped");
      await sleep(5000);
    }
  }
}

      for (const cond of exp.conditions) {
        const condRows = rows.filter((r) => r.experimentId === exp.id && r.condition === cond);

        if (exp.id === "pd") {
          const coop = condRows.filter((r) => r.choice === "C").length;

          const late = condRows.filter((r) => r.iteration > repsForSummary - 2 && r.choice === "C").length;
          const beliefRows = condRows.filter((r) => typeof r.belief === "number");

            ? beliefRows.filter((r) => r.extra.beliefCorrect === true).length
            : 0;

            experimentId: "pd",
            experimentLabel: "E. Repeated PD",
            condition: cond,
            n: condRows.length,
            metrics: {
              coopRate: pct(coop, condRows.length),
              earlyCoopRate: pct(early, Math.min(2, condRows.length)),
              lateCoopRate: pct(late, Math.min(2, condRows.length)),
              beliefAcc: beliefRows.length ? pct(belAcc, beliefRows.length) : 0,
              note: "finite-horizon rounds",

          });

          const contribs = condRows.map((r) => Number(r.extra.contribution)).filter((x) => Number.isFinite(x));

            experimentId: "pgg",
            experimentLabel: "F. Public Goods",
            condition: cond,

              meanContributionPct: +mean(contribs).toFixed(1),

              meanBelief: +mean(condRows.map((r) => Number(r.belief))).toFixed(2),
            },
          });

          const beliefs = condRows.map((r) => Number(r.belief)).filter((x) => Number.isFinite(x));

          const targetRate = pct(condRows.filter((r) => ["U", "L"].includes(String(r.choice))).length, condRows.length);

            experimentId: "mix",

            experimentId: "entry",
            experimentLabel: "H. Entry Deterrence",
            condition: cond,

              pHigh: +mean(condRows.map((r) => Number(r.extra.pHigh))).toFixed(2),

    }

  async function runPDObservation(cond: string, iter: number, persona: string, temperature: number): Promise<RawRow> {
    const flavor = flavorFor(iter);
    const priorRows = rowsForCond("pd", cond);

    const strat = cond.includes("TFT")
      ? "tit_for_tat"
      : cond.includes("AlwaysC")
      ? "always_cooperate"
      : cond.includes("AlwaysD")
      ? "always_defect"
      : "grim";

    const history = priorRows.slice(-8).length
      ? priorRows.slice(-8).map((r) => `R${r.iteration}: you=${r.choice} opp=${String(r.extra.opp ?? "C")}`).join(" | ")
      : "No prior rounds.";

    const prompt = `
Persona: ${persona}.
${flavor}
Prisoner's Dilemma. Condition: ${cond}
Round ${iter} of ${reps}.
History: ${history}

Payoffs:
CC=3 each
CD=0/5
DC=5/0
DD=1 each

Return strict JSON only:
{"beliefOppWillC": 0.0, "action": "C"}
`.trim();

    await rateLimitDelay();
    const rawText = await callLLM(prompt, temperature, () => stopRef.current);
    const parsed = parseJSON<any>(rawText) ?? {};
    const belief = typeof parsed.beliefOppWillC === "number" ? parsed.beliefOppWillC : 0.5;
    const action = typeof parsed.action === "string" ? parsed.action.toUpperCase() === "D" ? "D" : "C" : "C";

    const opp = strat === "always_cooperate"
      ? "C"
      : strat === "always_defect"
      ? "D"
      : strat === "grim"
      ? (priorRows.some((r) => r.choice === "D") || action === "D" ? "D" : "C")
      : priorRows.length === 0
      ? "C"
      : (priorRows[priorRows.length - 1].choice ?? "C");

    const payoff =
      action === "C" && opp === "C" ? 3 :
      action === "D" && opp === "C" ? 5 :
      action === "C" && opp === "D" ? 0 : 1;

    return {
      runId: `pd-${cond}-${iter}`,
      experimentId: "pd",
      experimentLabel: "E. Repeated PD",
      condition: cond,
      iteration: iter,
      persona,
      temperature,
      lambda,
      belief,
      choice: action,
      note: `payoff=${payoff}`,
      rawText,
      extra: {
        strat,
        opp,
        payoff,
        beliefCorrect: (belief >= 0.5 ? "C" : "D") === opp,
        brMatch: bestResponsePDAction() === action,
      },
    };
  }

  async function runCondition(exp: Experiment, cond: string) {
    const already = completedFor(exp.id, cond);
    const startIter = startIterationFor(exp.id, cond);

    if (already >= reps) {
      addLog(`[SKIP] ${cond} already complete (${already}/${reps})`);
      return;
    }

    if (startIter > 1) {
      addLog(`[RESUME] ${cond} continuing from iter ${startIter}/${reps}`);
    }

    for (let i = startIter; i <= reps; i++) {
      if (stopRef.current) break;
      await waitIfPaused();
      if (stopRef.current) break;

      const persona = personaFor(i);
      const temperature = tempFor(i);

      setCurrentExp(exp.id);
      setCurrentCond(cond);
      setCurrentIter(i);
      setStatus(`${exp.label} — ${cond}`);
      addLog(`${exp.id} | ${cond} | iter ${i}/${reps}`);

      try {
        const row =
          exp.id === "pd"
            ? await runPDObservation(cond, i, persona, temperature)
            : await runObservation(exp.id, cond, i, persona, temperature);

        appendRawRow(row);
        await sleep(80);
      } catch (err) {
        if (stopRef.current) break;
        addLog(`ERROR: ${err instanceof Error ? err.message : String(err)}`);
        addLog(`⏳ RETRYING SAME ITERATION ${i}...`);
        i--;
        await sleep(5000);
        continue;
      }
    }

    const done = completedFor(exp.id, cond);
    if (done >= reps) {
      addLog(`✅ ${cond} COMPLETE (${done}/${reps})`);
    } else {
      addLog(`⏸ ${cond} paused at ${done}/${reps}`);
    }
  }

  async function runExperiment(exp: Experiment) {
    if (stopRef.current) return;
    setCurrentExp(exp.id);

    for (const cond of exp.conditions) {
      if (stopRef.current) break;
      setCurrentCond(cond);
      await runCondition(exp, cond);
    }
  }

  async function runAll() {
    if (running) return;
    stopRef.current = false;
    pauseRef.current = false;
    setPaused(false);
    setStatus("Running remaining experiments...");
    setRunningState(true);

    try {
      setProgress((p) => ({ ...p, done: rawRowsRef.current.length, total: totalTarget }));
      for (const exp of EXPERIMENTS) {
        if (stopRef.current) break;
        await runExperiment(exp);
      }
      setStatus(stopRef.current ? "Stopped." : "Complete.");
    } catch (err) {
      setStatus(`Error: ${err instanceof Error ? err.message : String(err)}`);
      addLog(`ERROR: ${err instanceof Error ? err.message : String(err)}`);
    } finally {
      setCurrentExp("");
      setCurrentCond("");
      setCurrentIter(0);
      setRunningState(false);
    }
  }

  async function runOneExperiment(expId: ExperimentId) {
    if (running) return;
    const exp = EXPERIMENTS.find((e) => e.id === expId);
    if (!exp) return;

    stopRef.current = false;
    pauseRef.current = false;
    setPaused(false);
    setStatus(`Running ${exp.label}...`);
    setRunningState(true);

    try {
      await runExperiment(exp);
      setStatus(stopRef.current ? "Stopped." : `${exp.label} complete.`);
    } catch (err) {
      setStatus(`Error: ${err instanceof Error ? err.message : String(err)}`);
      addLog(`ERROR: ${err instanceof Error ? err.message : String(err)}`);
    } finally {
      setCurrentExp("");
      setCurrentCond("");
      setCurrentIter(0);
      setRunningState(false);
    }
  }

  function pause() {
    if (!running) return;
    pauseRef.current = true;
    setPaused(true);
    setStatus("Paused.");
    addLog("⏸ paused");
  }

  function resume() {
    if (!running) return;
    pauseRef.current = false;
    setPaused(false);
    setStatus("Resumed.");
    addLog("▶ resumed");
  }

  function stop() {
    stopRef.current = true;
    pauseRef.current = false;
    setPaused(false);
    setStatus("Stopping...");
    addLog("⏹ stop requested");
  }

  function downloadJSON(filename: string, obj: unknown) {
    const blob = new Blob([JSON.stringify(obj, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    a.click();
    URL.revokeObjectURL(url);
  }

        kappa,
        lambda0,
        lambda,
        reps,
        generatedAt: new Date().toISOString(),
      },
      rawRows: normalizeRows(rawRows),
      summaryRows,
      log,
    });
  }

              step="0.1"
              min="0.1"
              onChange={(e) => setLambda0(Number(e.target.value))}
              style={{ width: 80, marginLeft: 6 }}
            />
          </label>

            reps/condition{" "}
            <input
              type="number"
              value={reps}
              step="10"
              min="10"
              onChange={(e) => setReps(Math.max(10, Number(e.target.value) || 60))}
              style={{ width: 90, marginLeft: 6 }}
            />
          </label>

          <span style={{ color: "#fbbf24" }}>Current λ = {lambda.toFixed(3)}</span>

        </div>

        <div style={{ display: "flex", gap: 10, marginTop: 14, flexWrap: "wrap" }}>

          <button onClick={downloadAll} disabled={!rawRows.length} style={btnStyle}>Download All</button>

          return (

              }}
            >

              <div style={{ color: "#94a3b8", marginTop: 6, fontSize: 12 }}>
                {done}/{total} done

        })}
      </div>

      <div style={{ marginTop: 18, display: "grid", gridTemplateColumns: "1fr 0.9fr", gap: 14 }}>

                  <pre style={{ whiteSpace: "pre-wrap", margin: 0, color: "#fbbf24", fontSize: 12 }}>
                    {JSON.stringify(s.metrics, null, 2)}
                  </pre>
                </div>
              ))}
            </div>
          )}
        </div>

        <div style={{ background: "#0f172a", border: "1px solid #243046", borderRadius: 12, padding: 12 }}>
          <div style={{ fontWeight: 700, marginBottom: 10 }}>Log</div>
