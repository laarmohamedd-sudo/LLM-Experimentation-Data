function pick<T>(arr: T[]) {
  return arr[Math.floor(Math.random() * arr.length)];
}

function personaFor(i: number) {
  return PERSONAS[i % PERSONAS.length];
}

function flavorFor(i: number) {
  return FLAVORS[i % FLAVORS.length];
}

function tempFor(i: number) {
  const base = 0.9 + Math.random() * 0.1;
  const jitter = (i % 7) * 0.001;
  return Math.min(1, +(base + jitter).toFixed(3));
}

function sleep(ms: number) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function cleanJson(text: string) {
  return text.replace(/```json/gi, "").replace(/```/g, "").trim();
}

function parseJSON<T>(text: string): T | null {
  try {
    const cleaned = cleanJson(text);
    const start = cleaned.indexOf("{");
    const end = cleaned.lastIndexOf("}");
    const slice = start >= 0 && end > start ? cleaned.slice(start, end + 1) : cleaned;
    return JSON.parse(slice) as T;
  } catch {
    return null;
  }
}

function mean(xs: number[]) {
  const arr = xs.filter((x) => Number.isFinite(x));
  if (!arr.length) return NaN;
  return arr.reduce((a, b) => a + b, 0) / arr.length;
}

function sd(xs: number[]) {
  const arr = xs.filter((x) => Number.isFinite(x));
  if (arr.length < 2) return 0;
  const m = mean(arr);
  return Math.sqrt(arr.reduce((a, b) => a + (b - m) ** 2, 0) / (arr.length - 1));
}

function pct(num: number, den: number) {
  if (!den) return 0;
  return +(100 * num / den).toFixed(1);
}

function qreProbability(ua: number, ub: number, lambda: number) {
  return 1 / (1 + Math.exp(-lambda * (ua - ub)));
}

function bestResponseCoord(beliefOppA: number) {
  return beliefOppA >= 3 / 7 ? "A" : "B";
}

function bestResponseBertrand(mc: number, beliefPrice: number) {
  const grid = Array.from({ length: 21 }, (_, i) => mc + i);
  let bestP = grid[0];
  let bestProfit = -Infinity;

  for (const p of grid) {
    let profit = 0;
    if (p < mc) profit = -Infinity;
    else if (p < beliefPrice) profit = p - mc;
    else if (p === beliefPrice) profit = 0.5 * (p - mc);
    else profit = 0;

    if (profit > bestProfit || (profit === bestProfit && p < bestP)) {
      bestProfit = profit;
      bestP = p;
    }
  }

  return bestP;
}

function bestResponseAuction(v: number, beliefHighBid: number) {
  const grid = Array.from({ length: v + 1 }, (_, i) => i);
  let bestB = grid[0];
  let bestU = -Infinity;

  for (const b of grid) {
    let u = 0;
    if (b > v) u = -Infinity;
    else if (b > beliefHighBid) u = v - b;
    else if (b === beliefHighBid) u = 0.5 * (v - b);
    else u = 0;

    if (u > bestU || (u === bestU && b < bestB)) {
      bestU = u;
      bestB = b;
    }
  }

  return bestB;
}

function brMix(role: "row" | "col", belief: number) {
  if (role === "row") return belief > 1 / 3 ? "U" : "D";
  return belief > 2 / 3 ? "L" : "R";
}

async function callLLM(prompt: string, temperature: number) {
  const maxAttempts = 5;

  for (let attempt = 0; attempt < maxAttempts; attempt++) {
    const res = await fetch("http://localhost:3001/api/experiments/call", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ prompt, temperature }),
    });

    if (res.ok) {
      const data = (await res.json()) as { text?: string };
      return String(data.text ?? "");
    }

    if (res.status === 429) {
      const retryAfter = res.headers.get("retry-after");
      const waitMs = retryAfter ? Math.max(1000, Number(retryAfter) * 1000) : 2000 * (attempt + 1);
      await sleep(waitMs);
      continue;
    }

    throw new Error(`API ${res.status}: ${await res.text()}`);
  }

  throw new Error("API 429: retries exhausted");
}

Multiplier = ${m}.
${othersHint === null ? "You do not know others' contributions." : `Others contributed an average of $${othersHint} in the last round.`}

Return strict JSON only:
{"beliefOthersContrib": 0.0, "contribution": 0.0}
`.trim();

      const rawText = await callLLM(prompt, temperature);
      const parsed = parseJSON<any>(rawText) ?? {};
      const beliefOthers = typeof parsed.beliefOthersContrib === "number" ? parsed.beliefOthersContrib : 5;
      const contribution = typeof parsed.contribution === "number" ? Math.max(0, Math.min(10, parsed.contribution)) : 0;

      return {
        runId: `${exp}-${cond}-${iter}`,
        experimentId: exp,
        experimentLabel: "F. Public Goods",
        condition: cond,
        iteration: iter,
        persona,
        temperature,
        lambda,
        belief: beliefOthers,
        choice: String(contribution),
        note: `contrib=${contribution.toFixed(0)}`,
        rawText,
        extra: {
          N,
          m,
          othersHint,
          contribution,
          pctContribution: +(100 * contribution / 10).toFixed(1),
          free: contribution === 0,
        },
      };
    }

    if (exp === "mix") {
      const role = cond.startsWith("Row") ? "row" : "col";
      const frame = cond.split("-")[1] ?? "abstract";

      const prompt = role === "row"
        ? `
Persona: ${persona}.
${flavor}
Mixed 2x2 game (${frame} framing).
You are the Row player.

Payoffs:
(U,L)=2
(U,R)=0
(D,L)=0
(D,R)=1

Return strict JSON only:
{"beliefOppPlayL": 0.0, "choice": "U"}
`.trim()
        : `
Persona: ${persona}.
${flavor}
Mixed 2x2 game (${frame} framing).
You are the Column player.

Payoffs:
(U,L)=1
(U,R)=0
(D,L)=0
(D,R)=2

Return strict JSON only:
{"beliefOppPlayU": 0.0, "choice": "L"}
`.trim();

      const rawText = await callLLM(prompt, temperature);
      const parsed = parseJSON<any>(rawText) ?? {};

      const belief = role === "row"
        ? (typeof parsed.beliefOppPlayL === "number" ? parsed.beliefOppPlayL : 0.5)
        : (typeof parsed.beliefOppPlayU === "number" ? parsed.beliefOppPlayU : 0.5);

      const choice = typeof parsed.choice === "string" ? parsed.choice.toUpperCase() : null;
      const br = brMix(role, belief);
      const pA = role === "row"
        ? qreProbability(2 * belief, 1 * (1 - belief), lambda)
        : qreProbability(1 * belief, 2 * (1 - belief), lambda);

      return {
        runId: `${exp}-${cond}-${iter}`,
        experimentId: exp,
        experimentLabel: "G. Mixed 2x2",
        condition: cond,
        iteration: iter,
        persona,
        temperature,
        lambda,
        belief,
        choice,
        note: choice ? `choice=${choice}` : "mix",
        rawText,
        extra: {
          role,
          frame,
          pA,
          br,
          brMatch: choice != null ? choice === br : null,
        },
      };
    }

    if (exp === "entry") {
      const frame = cond.toLowerCase();

      const prompt = `
Persona: ${persona}.
${flavor}
Entry-deterrence game (${frame} framing).

Incumbent chooses High or Low.
Entrant chooses Enter or Stay out.

Payoffs:
If incumbent chose High:
- Enter => entrant 3
- Stay out => entrant 0
If incumbent chose Low:
- Enter => entrant -1
- Stay out => entrant 0

Return strict JSON only:
{"pHigh": 0.0, "pLow": 0.0, "commitment": "High", "entrant": "Enter"}
`.trim();

      const rawText = await callLLM(prompt, temperature);
      const parsed = parseJSON<any>(rawText) ?? {};
      const pHigh = typeof parsed.pHigh === "number" ? parsed.pHigh : 0.85;
      const pLow = typeof parsed.pLow === "number" ? parsed.pLow : 0.2;
      const commitment = typeof parsed.commitment === "string" && parsed.commitment.toLowerCase().includes("low") ? "Low" : "High";
      const entrant = typeof parsed.entrant === "string" && parsed.entrant.toLowerCase().includes("enter") ? "Enter" : "Stay out";
      const brCommit = 4 - 2 * pHigh >= 3 - 2 * pLow ? "High" : "Low";
      const brEntrant = commitment === "High" ? "Enter" : "Stay out";

      return {
        runId: `${exp}-${cond}-${iter}`,
        experimentId: exp,
        experimentLabel: "H. Entry Deterrence",
        condition: cond,
        iteration: iter,
        persona,
        temperature,
        lambda,
        belief: pHigh,
        choice: commitment,
        note: `commit=${commitment}`,
        rawText,
        extra: {
          pHigh,
          pLow,
          brCommit,
          brEntrant,
          brMatchCommit: commitment === brCommit,
          brMatchEntrant: entrant === brEntrant,
          entrant,
        },
      };
    }

    if (exp === "iia") {
      const domain = cond.split(":")[0].toLowerCase();
      const decoy = cond.includes("Dec");
      const tgt = cond.includes("DecA") ? "A" : cond.includes("DecB") ? "B" : null;

      const prompt = domain === "prod"
        ? `
Persona: ${persona}.
${flavor}
Choose one product.

${decoy ? (tgt === "A"

            : "B: Quality 5/10, Price 30\nC: Quality 4/10, Price 35")

Return strict JSON only:
{"choice": "A"}
`.trim()
        : domain === "job"
        ? `
Persona: ${persona}.
${flavor}
Choose one candidate.

A: Experience 7/10, Salary 90
${decoy ? (tgt === "A"
            ? "B: Experience 5/10, Salary 60\nC: Experience 6/10, Salary 95"
            : "B: Experience 5/10, Salary 60\nC: Experience 4/10, Salary 65")
          : "B: Experience 5/10, Salary 60"}

Return strict JSON only:
{"choice": "A"}
`.trim()
        : `
Persona: ${persona}.
${flavor}
Choose one investment.

A: Return 7%, Risk 5/10
${decoy ? (tgt === "A"
            ? "B: Return 4%, Risk 2/10\nC: Return 6%, Risk 6/10"
            : "B: Return 4%, Risk 2/10\nC: Return 3%, Risk 3/10")
          : "B: Return 4%, Risk 2/10"}

Return strict JSON only:
{"choice": "A"}
`.trim();

      const rawText = await callLLM(prompt, temperature);
      const parsed = parseJSON<any>(rawText) ?? {};
      const choice = typeof parsed.choice === "string" ? parsed.choice.toUpperCase() : null;

      return {
        runId: `${exp}-${cond}-${iter}`,
        experimentId: exp,
        experimentLabel: "I. IIA / Decoy Sensitivity",
        condition: cond,
        iteration: iter,
        persona,
        temperature,
        lambda,
        belief: null,
        choice,
        note: choice ? `choice=${choice}` : "iia",
        rawText,
        extra: {
          domain,
          decoy,
          tgt,
        },
      };
    }

    throw new Error(`Unknown experiment: ${exp}`);
  }

  async function runCondition(exp: Experiment, cond: string) {
    const conditionRows: RawRow[] = [];
    const pdHistory: { llm: string; opp: string }[] = [];
    let everDefected = false;

    for (let i = 1; i <= reps; i++) {
      if (stopRef.current) break;
      await waitIfPaused();
      if (stopRef.current) break;

      const persona = personaFor(i);
      const temperature = tempFor(i);

      setCurrentIter(i);
      addLog(`${exp.id} | ${cond} | iter ${i}/${reps}`);

      try {
        if (exp.id === "pd") {
          const strat = cond.includes("TFT")
            ? "tit_for_tat"
            : cond.includes("AlwaysC")
            ? "always_cooperate"
            : cond.includes("AlwaysD")
            ? "always_defect"
            : "grim";

          const oppMove = () => {
            if (strat === "always_cooperate") return "C";
            if (strat === "always_defect") return "D";
            if (strat === "grim") return everDefected ? "D" : "C";
            return pdHistory.length === 0 ? "C" : pdHistory[pdHistory.length - 1].llm;
          };

          const hist = pdHistory.slice(-8).length
            ? pdHistory.slice(-8).map((h, idx) => `R${Math.max(1, pdHistory.length - 7) + idx}: you=${h.llm} opp=${h.opp}`).join(" | ")
            : "No prior rounds.";

          const prompt = `
Persona: ${persona}.
${flavorFor(i)}
Prisoner's Dilemma. Condition: ${cond}
Round ${i} of ${reps}.
History: ${hist}

Payoffs:
CC=3 each
CD=0/5
DC=5/0
DD=1 each

Return strict JSON only:
{"beliefOppWillC": 0.0, "action": "C"}
`.trim();

          await rateLimitDelay();
          const rawText = await callLLM(prompt, temperature);
          const parsed = parseJSON<any>(rawText) ?? {};
          const belief = typeof parsed.beliefOppWillC === "number" ? parsed.beliefOppWillC : 0.5;
          const action = typeof parsed.action === "string" ? parsed.action.toUpperCase() === "D" ? "D" : "C" : "C";

          if (action === "D") everDefected = true;
          const opp = oppMove();
          pdHistory.push({ llm: action, opp });

          const payoff =
            action === "C" && opp === "C" ? 3 :
            action === "D" && opp === "C" ? 5 :
            action === "C" && opp === "D" ? 0 : 1;

          const row: RawRow = {
            runId: `${exp.id}-${cond}-${i}`,
            experimentId: exp.id,
            experimentLabel: exp.label,
            condition: cond,
            iteration: i,
            persona,
            temperature,
            lambda,
            belief,
            choice: action,
            note: `payoff=${payoff}`,
            rawText,
            extra: {
              strat,
              opp,
              payoff,
              beliefCorrect: (belief >= 0.5 ? "C" : "D") === opp,
              brMatch: action === "D",
            },
          };

          conditionRows.push(row);
          setRawRows((prev) => [...prev, row]);
          upsertSummaryRow(exp.id, cond, [...conditionRows]);
          setProgress((p) => ({ ...p, done: p.done + 1 }));
          await sleep(80);
          continue;
        }

        const row = await runObservation(exp.id, cond, i, persona, temperature);
        conditionRows.push(row);
        setRawRows((prev) => [...prev, row]);
        upsertSummaryRow(exp.id, cond, [...conditionRows]);
        setProgress((p) => ({ ...p, done: p.done + 1 }));
        await sleep(80);
      } catch (err) {
        addLog(`ERROR: ${err instanceof Error ? err.message : String(err)}`);
        break;
      }
    }

    if (conditionRows.length) {
      upsertSummaryRow(exp.id, cond, conditionRows);
    }
  }

  async function runExperiment(exp: Experiment) {
    if (stopRef.current) return;
    setCurrentExp(exp.id);

    for (const cond of exp.conditions) {
      if (stopRef.current) break;
      setCurrentCond(cond);
      setStatus(`${exp.label} — ${cond}`);
      await runCondition(exp, cond);
    }
  }

  async function runAll() {
    if (running) return;
    stopRef.current = false;
    pauseRef.current = false;
    setPaused(false);
    setRawRows([]);
    setSummaryRows([]);
    setLog([]);
    setProgress({ done: 0, total: totalTarget });
    setCurrentExp("");
    setCurrentCond("");
    setCurrentIter(0);
    setStatus("Running all experiments...");
    setRunningState(true);

    try {
      for (const exp of EXPERIMENTS) {
        if (stopRef.current) break;
        await runExperiment(exp);
      }
      setStatus(stopRef.current ? "Stopped." : "Complete.");
    } catch (err) {
      setStatus(`Error: ${err instanceof Error ? err.message : String(err)}`);
      addLog(`ERROR: ${err instanceof Error ? err.message : String(err)}`);
    } finally {
      setCurrentExp("");
      setCurrentCond("");
      setCurrentIter(0);
      setRunningState(false);
    }
  }

  async function runOneExperiment(expId: ExperimentId) {
    if (running) return;
    const exp = EXPERIMENTS.find((e) => e.id === expId);
    if (!exp) return;

    stopRef.current = false;
    pauseRef.current = false;
    setPaused(false);
    setRawRows([]);
    setSummaryRows([]);
    setLog([]);
    setProgress({ done: 0, total: exp.conditions.length * reps });
    setCurrentExp(expId);
    setCurrentCond("");
    setCurrentIter(0);
    setStatus(`Running ${exp.label}...`);
    setRunningState(true);

    try {
      await runExperiment(exp);
      setStatus(stopRef.current ? "Stopped." : `${exp.label} complete.`);
    } catch (err) {
      setStatus(`Error: ${err instanceof Error ? err.message : String(err)}`);
      addLog(`ERROR: ${err instanceof Error ? err.message : String(err)}`);
    } finally {
      setCurrentExp("");
      setCurrentCond("");
      setCurrentIter(0);
      setRunningState(false);
    }
  }

  function pause() {
    if (!running) return;
    pauseRef.current = true;
    setPaused(true);
    setStatus("Paused.");
    addLog("⏸ paused");
  }

  function resume() {
    if (!running) return;
    pauseRef.current = false;
    setPaused(false);
    setStatus("Resumed.");
    addLog("▶ resumed");
  }

  function stop() {
    stopRef.current = true;
    pauseRef.current = false;
    setPaused(false);
    setStatus("Stopping...");
    addLog("⏹ stop requested");
  }

  function downloadJSON(filename: string, obj: unknown) {
    const blob = new Blob([JSON.stringify(obj, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    a.click();
    URL.revokeObjectURL(url);
  }

    downloadJSON("artificial-rationality-raw-and-summary.json", {
      meta: {
        kappa,
        lambda0,
        lambda,
        reps,
        generatedAt: new Date().toISOString(),
      },
      rawRows,
      summaryRows,
    });
  }

    stopRef.current = true;
    pauseRef.current = false;
    setPaused(false);
    setRunningState(false);
    setStatus("Cleared.");
    setRawRows([]);
    setSummaryRows([]);
    setLog([]);
    setProgress({ done: 0, total: 0 });
    setCurrentExp("");
    setCurrentCond("");
    setCurrentIter(0);
  }

  const latestRows = rawRows.slice(-120);

      <div style={{ borderBottom: "1px solid #243046", paddingBottom: 12, marginBottom: 16 }}>

        <div style={{ marginTop: 6, color: "#94a3b8", fontSize: 13 }}>
          Beliefs → Actions → Best-response check. 9 experiments, sequential runs, raw rows separate from summaries. λ = λ₀ × κ = {lambda.toFixed(3)}
        </div>

        <div style={{ display: "flex", gap: 12, marginTop: 14, flexWrap: "wrap", alignItems: "center" }}>
          <label>
            κ{" "}
            <input
              type="number"
              value={kappa}
              step="0.1"
              min="0.1"
              onChange={(e) => setKappa(Number(e.target.value))}
              style={{ width: 80, marginLeft: 6 }}
            />
          </label>

          <label>
            λ₀{" "}
            <input
              type="number"
              value={lambda0}
              step="0.1"
              min="0.1"
              onChange={(e) => setLambda0(Number(e.target.value))}
              style={{ width: 80, marginLeft: 6 }}
            />
          </label>

          <label>
            reps/condition{" "}
            <input
              type="number"
              value={reps}
              step="50"
              min="50"
              onChange={(e) => setReps(Math.max(50, Number(e.target.value) || 60))}
              style={{ width: 90, marginLeft: 6 }}
            />
          </label>

          <span style={{ color: "#fbbf24" }}>Current λ = {lambda.toFixed(3)}</span>
          <span style={{ color: "#86efac" }}>{status}</span>
          <span style={{ color: "#93c5fd" }}>
            {progress.done}/{progress.total || totalTarget} ({pct(progress.done, progress.total || totalTarget)}%)
          </span>
          {currentExp && (
            <span style={{ color: "#c084fc" }}>
              {currentExp} {currentCond ? `— ${currentCond}` : ""} {currentIter ? `— iter ${currentIter}` : ""}
            </span>
          )}
        </div>

        <div style={{ display: "flex", gap: 10, marginTop: 14, flexWrap: "wrap" }}>
          <button onClick={runAll} disabled={running} style={btnStyle}>Run All</button>
          <button onClick={pause} disabled={!running || paused} style={btnStyle}>Pause</button>

      </div>

      <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit, minmax(220px, 1fr))", gap: 12 }}>
        {EXPERIMENTS.map((e) => (
          <button
            key={e.id}
            disabled={running}

              color: "white",

          </button>
        ))}
      </div>

      <div style={{ marginTop: 18, display: "grid", gridTemplateColumns: "1fr 0.9fr", gap: 14 }}>

          {summaryRows.length === 0 ? (

          ) : (
            <div style={{ display: "grid", gap: 10 }}>
              {summaryRows.map((s) => (
                <div key={`${s.experimentId}-${s.condition}`} style={{ border: "1px solid #1f2a3d", borderRadius: 10, padding: 10 }}>
                  <div style={{ fontWeight: 700, marginBottom: 6 }}>{s.experimentLabel}</div>

      <div style={{ marginTop: 18, background: "#0f172a", border: "1px solid #243046", borderRadius: 12, padding: 12 }}>
        <div style={{ fontWeight: 700, marginBottom: 10 }}>Latest raw rows (last 120 only)</div>
        {latestRows.length === 0 ? (
          <div style={{ color: "#94a3b8" }}>No raw rows yet.</div>

          <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 12 }}>

                  <td style={td}>{r.experimentLabel}</td>
                  <td style={td}>{r.condition}</td>
                  <td style={td}>{r.iteration}</td>
                  <td style={td}>{r.belief == null ? "—" : r.belief.toFixed(3)}</td>
                  <td style={td}>{r.choice ?? "—"}</td>
                  <td style={{ ...td, color: "#fbbf24" }}>{r.note}</td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>

      <div style={{ marginTop: 14, color: "#94a3b8", fontSize: 12 }}>
        Raw rows are one draw each. Summary rows are computed and updated as the condition runs.
      </div>
    </div>
  );
}
